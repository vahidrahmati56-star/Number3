<script type="module">
import * as THREE from 'three';
import { EffectComposer } from 'three/addons/postprocessing/EffectComposer.js';
import { RenderPass } from 'three/addons/postprocessing/RenderPass.js';
import { UnrealBloomPass } from 'three/addons/postprocessing/UnrealBloomPass.js';

/* ═══════════ SETUP ═══════════ */
const scene = new THREE.Scene();
scene.background = new THREE.Color(0x0a0e1a);
scene.fog = new THREE.FogExp2(0x1a1525, 0.012);

const camera = new THREE.PerspectiveCamera(55, innerWidth/innerHeight, 0.1, 400);

const renderer = new THREE.WebGLRenderer({antialias:true, powerPreference:'high-performance'});
renderer.setPixelRatio(Math.min(devicePixelRatio, 2));
renderer.setSize(innerWidth, innerHeight);
renderer.shadowMap.enabled = true;
renderer.shadowMap.type = THREE.PCFSoftShadowMap;
renderer.toneMapping = THREE.ACESFilmicToneMapping;
renderer.toneMappingExposure = 1.15;
renderer.outputColorSpace = THREE.SRGBColorSpace;
document.body.appendChild(renderer.domElement);

const composer = new EffectComposer(renderer);
composer.addPass(new RenderPass(scene, camera));
const bloom = new UnrealBloomPass(new THREE.Vector2(innerWidth, innerHeight), 0.55, 0.5, 0.82);
composer.addPass(bloom);

addEventListener('resize', () => {
  camera.aspect = innerWidth / innerHeight;
  camera.updateProjectionMatrix();
  renderer.setSize(innerWidth, innerHeight);
  composer.setSize(innerWidth, innerHeight);
});

/* ═══════════ LIGHTS ═══════════ */
const sun = new THREE.DirectionalLight(0xffe0b0, 2.2);
sun.position.set(40, 70, 30);
sun.castShadow = true;
sun.shadow.mapSize.set(2048, 2048);
sun.shadow.camera.left = -40;
sun.shadow.camera.right = 40;
sun.shadow.camera.top = 40;
sun.shadow.camera.bottom = -40;
sun.shadow.camera.near = 1;
sun.shadow.camera.far = 150;
sun.shadow.bias = -0.0004;
sun.shadow.normalBias = 0.02;
scene.add(sun);
scene.add(sun.target);

const hemi = new THREE.HemisphereLight(0x88aaff, 0x4a2a10, 0.55);
scene.add(hemi);

const ambient = new THREE.AmbientLight(0x405070, 0.4);
scene.add(ambient);

/* ═══════════ MATERIALS ═══════════ */
const MAT = {
  skin: new THREE.MeshStandardMaterial({color:0xd9a878, roughness:0.72}),
  tunic: new THREE.MeshStandardMaterial({color:0xf5e6c8, roughness:0.85}),
  belt: new THREE.MeshStandardMaterial({color:0x6d3510, roughness:0.75}),
  gold: new THREE.MeshStandardMaterial({color:0xffcc44, roughness:0.28, metalness:0.85, emissive:0x442200, emissiveIntensity:0.15}),
  steel: new THREE.MeshStandardMaterial({color:0xdde5ee, roughness:0.15, metalness:0.95, emissive:0x223344, emissiveIntensity:0.15}),
  hair: new THREE.MeshStandardMaterial({color:0x1a0e08, roughness:0.9}),
  dark: new THREE.MeshStandardMaterial({color:0x2a1a10, roughness:0.85}),
  red: new THREE.MeshStandardMaterial({color:0x9b2226, roughness:0.7, emissive:0x330000, emissiveIntensity:0.3}),
  stone: new THREE.MeshStandardMaterial({color:0xa89070, roughness:0.95}),
  stoneDark: new THREE.MeshStandardMaterial({color:0x6d5540, roughness:0.98}),
  sand: new THREE.MeshStandardMaterial({color:0xc9a876, roughness:0.95}),
  marble: new THREE.MeshStandardMaterial({color:0xe8dcc0, roughness:0.5, metalness:0.1}),
  wood: new THREE.MeshStandardMaterial({color:0x5a3018, roughness:0.9}),
  flame: new THREE.MeshBasicMaterial({color:0xffaa44, transparent:true, opacity:0.9})
};

/* ═══════════ CHARACTER BUILDER ═══════════ */
function buildHumanoid(opts = {}){
  const c = Object.assign({
    skin: MAT.skin, tunic: MAT.tunic, belt: MAT.belt,
    hair: MAT.hair, gold: MAT.gold, steel: MAT.steel,
    hasSword: true, hasCape: false
  }, opts);
  
  const root = new THREE.Group();
  
  const pelvis = new THREE.Group();
  pelvis.position.y = 0.9;
  root.add(pelvis);
  
  const torso = new THREE.Mesh(new THREE.CapsuleGeometry(0.3, 0.6, 8, 12), c.tunic);
  torso.position.y = 0.35;
  torso.castShadow = true;
  pelvis.add(torso);
  
  const chest = new THREE.Mesh(new THREE.BoxGeometry(0.52, 0.34, 0.28), c.tunic);
  chest.position.y = 0.58;
  chest.castShadow = true;
  pelvis.add(chest);
  
  const beltMesh = new THREE.Mesh(new THREE.CylinderGeometry(0.36, 0.36, 0.14, 12), c.belt);
  beltMesh.position.y = 0.06;
  beltMesh.castShadow = true;
  pelvis.add(beltMesh);
  
  const neck = new THREE.Mesh(new THREE.CylinderGeometry(0.09, 0.11, 0.15, 8), c.skin);
  neck.position.y = 0.78;
  pelvis.add(neck);
  
  const head = new THREE.Mesh(new THREE.SphereGeometry(0.22, 20, 14), c.skin);
  head.position.y = 1.0;
  head.scale.set(1, 1.18, 0.95);
  head.castShadow = true;
  pelvis.add(head);
  
  const hairMesh = new THREE.Mesh(new THREE.SphereGeometry(0.23, 20, 14), c.hair);
  hairMesh.position.y = 1.05;
  hairMesh.scale.set(1.02, 0.85, 1.0);
  pelvis.add(hairMesh);
  
  // turban
  const turban = new THREE.Mesh(new THREE.TorusGeometry(0.21, 0.09, 10, 20), c.tunic);
  turban.position.y = 1.13;
  turban.rotation.x = Math.PI/2;
  turban.castShadow = true;
  pelvis.add(turban);
  
  const turbanTop = new THREE.Mesh(new THREE.SphereGeometry(0.16, 12, 8), c.tunic);
  turbanTop.position.y = 1.2;
  turbanTop.scale.y = 0.6;
  pelvis.add(turbanTop);
  
  const turbanJewel = new THREE.Mesh(new THREE.SphereGeometry(0.05, 8, 6), c.gold);
  turbanJewel.position.set(0, 1.13, 0.22);
  pelvis.add(turbanJewel);
  
  // arms
  const shoulders = [], elbows = [];
  for (let i = 0; i < 2; i++){
    const side = i === 0 ? -1 : 1;
    const shoulder = new THREE.Group();
    shoulder.position.set(side * 0.36, 0.68, 0);
    pelvis.add(shoulder);
    shoulders.push(shoulder);
    
    const upperArm = new THREE.Mesh(new THREE.CapsuleGeometry(0.085, 0.34, 6, 8), c.skin);
    upperArm.position.y = -0.22;
    upperArm.castShadow = true;
    shoulder.add(upperArm);
    
    const sleeve = new THREE.Mesh(new THREE.CylinderGeometry(0.13, 0.11, 0.22, 10), c.tunic);
    sleeve.position.y = -0.06;
    sleeve.castShadow = true;
    shoulder.add(sleeve);
    
    const elbow = new THREE.Group();
    elbow.position.y = -0.44;
    shoulder.add(elbow);
    elbows.push(elbow);
    
    const forearm = new THREE.Mesh(new THREE.CapsuleGeometry(0.075, 0.34, 6, 8), c.skin);
    forearm.position.y = -0.22;
    forearm.castShadow = true;
    elbow.add(forearm);
    
    const hand = new THREE.Mesh(new THREE.SphereGeometry(0.095, 10, 8), c.skin);
    hand.position.y = -0.44;
    hand.castShadow = true;
    elbow.add(hand);
    
    if (c.hasSword && side === 1){
      const sword = new THREE.Group();
      const blade = new THREE.Mesh(new THREE.BoxGeometry(0.055, 0.95, 0.014), c.steel);
      blade.position.y = 0.53;
      blade.castShadow = true;
      sword.add(blade);
      const tip = new THREE.Mesh(new THREE.ConeGeometry(0.04, 0.12, 4), c.steel);
      tip.position.y = 1.05;
      tip.rotation.y = Math.PI/4;
      sword.add(tip);
      const guard = new THREE.Mesh(new THREE.BoxGeometry(0.28, 0.04, 0.06), c.gold);
      guard.position.y = 0.05;
      guard.castShadow = true;
      sword.add(guard);
      const hilt = new THREE.Mesh(new THREE.CylinderGeometry(0.028, 0.028, 0.18, 8), c.belt);
      hilt.position.y = -0.06;
      sword.add(hilt);
      const pommel = new THREE.Mesh(new THREE.SphereGeometry(0.045, 10, 8), c.gold);
      pommel.position.y = -0.17;
      sword.add(pommel);
      
      sword.position.y = -0.5;
      sword.rotation.x = Math.PI * 0.55;
      hand.add(sword);
    }
  }
  
  // legs
  const hips = [], knees = [];
  for (let i = 0; i < 2; i++){
    const side = i === 0 ? -1 : 1;
    const hip = new THREE.Group();
    hip.position.set(side * 0.16, -0.02, 0);
    pelvis.add(hip);
    hips.push(hip);
    
    const upperLeg = new THREE.Mesh(new THREE.CapsuleGeometry(0.1, 0.4, 6, 8), c.tunic);
    upperLeg.position.y = -0.26;
    upperLeg.castShadow = true;
    hip.add(upperLeg);
    
    const knee = new THREE.Group();
    knee.position.y = -0.55;
    hip.add(knee);
    knees.push(knee);
    
    const lowerLeg = new THREE.Mesh(new THREE.CapsuleGeometry(0.078, 0.4, 6, 8), c.skin);
    lowerLeg.position.y = -0.24;
    lowerLeg.castShadow = true;
    knee.add(lowerLeg);
    
    const foot = new THREE.Mesh(new THREE.BoxGeometry(0.14, 0.1, 0.26), c.belt);
    foot.position.set(0, -0.46, 0.06);
    foot.castShadow = true;
    knee.add(foot);
  }
  
  // cape
  let cape = null;
  if (c.hasCape){
    cape = new THREE.Group();
    cape.position.set(0, 0.75, -0.16);
    pelvis.add(cape);
    for (let i = 0; i < 5; i++){
      const seg = new THREE.Mesh(
        new THREE.BoxGeometry(0.5 - i*0.06, 0.24, 0.02),
        new THREE.MeshStandardMaterial({color:0x8b0000, roughness:0.7, side:THREE.DoubleSide})
      );
      seg.position.y = -i * 0.22;
      seg.castShadow = true;
      cape.add(seg);
    }
  }
  
  return {root, pelvis, torso, chest, head, shoulders, elbows, hips, knees, cape, sword: shoulders[1].children[shoulders[1].children.length-1]?.children[0]?.children?.find?.(x=>x)};
}

/* ═══════════ ANIMATION ═══════════ */
function poseCharacter(rig, pose, t, speed = 1){
  const {pelvis, torso, head, shoulders, elbows, hips, knees, cape} = rig;
  
  // baseline
  pelvis.position.y = 0.9;
  pelvis.rotation.set(0, 0, 0);
  torso.rotation.set(0, 0, 0);
  head.rotation.set(0, 0, 0);
  shoulders[0].rotation.set(0, 0, 0);
  shoulders[1].rotation.set(0, 0, 0);
  elbows[0].rotation.set(0, 0, 0);
  elbows[1].rotation.set(0, 0, 0);
  hips[0].rotation.set(0, 0, 0);
  hips[1].rotation.set(0, 0, 0);
  knees[0].rotation.set(0, 0, 0);
  knees[1].rotation.set(0, 0, 0);
  
  const breath = Math.sin(t * 2) * 0.015;
  
  if (pose === 'idle'){
    pelvis.position.y = 0.9 + breath;
    shoulders[0].rotation.x = 0.08;
    shoulders[1].rotation.x = 0.08;
    shoulders[0].rotation.z = 0.12;
    shoulders[1].rotation.z = -0.12;
    elbows[0].rotation.x = -0.25;
    elbows[1].rotation.x = -0.3;
    head.rotation.y = Math.sin(t * 0.4) * 0.12;
    torso.rotation.y = Math.sin(t * 0.6) * 0.05;
  }
  else if (pose === 'walk' || pose === 'run'){
    const freq = pose === 'run' ? 9 : 5;
    const amp = pose === 'run' ? 1.05 : 0.65;
    const phase = t * freq;
    
    hips[0].rotation.x = Math.sin(phase) * amp;
    hips[1].rotation.x = Math.sin(phase + Math.PI) * amp;
    knees[0].rotation.x = Math.max(0, -Math.sin(phase + 0.5)) * amp * 1.3;
    knees[1].rotation.x = Math.max(0, -Math.sin(phase + Math.PI + 0.5)) * amp * 1.3;
    
    shoulders[0].rotation.x = Math.sin(phase + Math.PI) * amp * 0.85;
    shoulders[1].rotation.x = Math.sin(phase) * amp * 0.85;
    shoulders[0].rotation.z = 0.15;
    shoulders[1].rotation.z = -0.15;
    elbows[0].rotation.x = -0.4 - Math.abs(Math.sin(phase + Math.PI)) * 0.4;
    elbows[1].rotation.x = -0.4 - Math.abs(Math.sin(phase)) * 0.4;
    
    pelvis.position.y = 0.9 + Math.abs(Math.sin(phase)) * 0.06 * amp;
    pelvis.rotation.z = Math.sin(phase) * 0.06 * amp;
    pelvis.rotation.x = pose === 'run' ? -0.15 : -0.05;
    torso.rotation.y = -Math.sin(phase) * 0.12 * amp;
    head.rotation.x = pose === 'run' ? 0.1 : 0;
  }
  else if (pose === 'jump'){
    hips[0].rotation.x = -0.7;
    hips[1].rotation.x = 0.5;
    knees[0].rotation.x = 1.0;
    knees[1].rotation.x = 0.7;
    shoulders[0].rotation.x = -2.2;
    shoulders[1].rotation.x = -2.0;
    shoulders[0].rotation.z = 0.4;
    shoulders[1].rotation.z = -0.4;
    elbows[0].rotation.x = -0.3;
    elbows[1].rotation.x = -0.5;
    pelvis.rotation.x = -0.2;
    head.rotation.x = -0.1;
  }
  else if (pose === 'fall'){
    hips[0].rotation.x = 0.4;
    hips[1].rotation.x = -0.4;
    knees[0].rotation.x = 0.5;
    knees[1].rotation.x = 0.5;
    shoulders[0].rotation.x = -1.8;
    shoulders[1].rotation.x = -1.6;
    shoulders[0].rotation.z = 0.6;
    shoulders[1].rotation.z = -0.6;
    elbows[0].rotation.x = -0.6;
    elbows[1].rotation.x = -0.6;
    pelvis.rotation.x = 0.1;
  }
  else if (pose === 'wallrun'){
    const dir = speed >= 0 ? 1 : -1;
    pelvis.rotation.z = 0.35 * dir;
    hips[0].rotation.x = Math.sin(t * 14) * 1.0;
    hips[1].rotation.x = Math.sin(t * 14 + Math.PI) * 1.0;
    knees[0].rotation.x = Math.abs(Math.sin(t * 14)) * 1.2;
    knees[1].rotation.x = Math.abs(Math.sin(t * 14 + Math.PI)) * 1.2;
    shoulders[0].rotation.x = Math.sin(t * 14 + Math.PI) * 0.85;
    shoulders[1].rotation.x = Math.sin(t * 14) * 0.85;
    shoulders[0].rotation.z = 0.5;
    shoulders[1].rotation.z = -0.5;
    elbows[0].rotation.x = -0.5;
    elbows[1].rotation.x = -0.5;
    pelvis.rotation.x = -0.25;
    torso.rotation.z = -0.15 * dir;
    head.rotation.z = -0.2 * dir;
  }
  else if (pose === 'attack'){
    const p = Math.min(1, t * 5);
    const swing = Math.sin(p * Math.PI);
    const windup = Math.max(0, 1 - p * 3);
    shoulders[1].rotation.x = -2.6 * swing - 0.5 * windup;
    shoulders[1].rotation.z = -1.0 * swing;
    shoulders[1].rotation.y = -0.5 * swing;
    elbows[1].rotation.x = -0.15 - 0.8 * windup;
    shoulders[0].rotation.x = 0.4 * swing - 0.3 * windup;
    shoulders[0].rotation.z = 0.4;
    torso.rotation.y = -0.55 * swing;
    torso.rotation.x = 0.15 * swing;
    pelvis.rotation.y = -0.25 * swing;
  }
  else if (pose === 'hurt'){
    pelvis.rotation.x = 0.25;
    shoulders[0].rotation.x = 0.5;
    shoulders[1].rotation.x = 0.5;
    shoulders[0].rotation.z = -0.3;
    shoulders[1].rotation.z = 0.3;
    elbows[0].rotation.x = -1.4;
    elbows[1].rotation.x = -1.4;
    head.rotation.x = 0.3;
  }
  else if (pose === 'wallslide'){
    pelvis.rotation.z = 0.2;
    hips[0].rotation.x = 0.3;
    hips[1].rotation.x = -0.2;
    knees[0].rotation.x = 0.6;
    knees[1].rotation.x = 0.4;
    shoulders[0].rotation.x = -1.0;
    shoulders[1].rotation.x = -1.0;
    elbows[0].rotation.x = -0.5;
    elbows[1].rotation.x = -0.5;
  }
  
  // cape simulation
  if (cape){

/* ═══════════ PHYSICS ═══════════ */
function getAABB(pos){
  return {
    min: {x: pos.x - PLAYER_RADIUS, y: pos.y, z: pos.z - PLAYER_RADIUS},
    max: {x: pos.x + PLAYER_RADIUS, y: pos.y + PLAYER_HEIGHT, z: pos.z + PLAYER_RADIUS}
  };
}
function overlap(a, b){
  return a.min.x < b.max.x && a.max.x > b.min.x &&
         a.min.y < b.max.y && a.max.y > b.min.y &&
         a.min.z < b.max.z && a.max.z > b.min.z;
}

function resolveCollisions(axis, delta){
  const box = getAABB(player.pos);
  let hitWall = null;
  for (const c of colliders){
    if (!overlap(box, c)) continue;
    if (axis === 'y'){
      if (delta < 0){
        player.pos.y = c.max.y;
        player.vel.y = 0;
        player.onGround = true;
        player.coyote = 8;
        player.doubleJumpUsed = false;
      } else if (delta > 0){
        player.pos.y = c.min.y - PLAYER_HEIGHT;
        player.vel.y = 0;
      }
    } else if (axis === 'x'){
      if (delta > 0){
        player.pos.x = c.min.x - PLAYER_RADIUS;
        hitWall = new THREE.Vector3(-1, 0, 0);
      } else {
        player.pos.x = c.max.x + PLAYER_RADIUS;
        hitWall = new THREE.Vector3(1, 0, 0);
      }
      player.vel.x = 0;
    } else if (axis === 'z'){
      if (delta > 0){
        player.pos.z = c.min.z - PLAYER_RADIUS;
        hitWall = new THREE.Vector3(0, 0, -1);
      } else {
        player.pos.z = c.max.z + PLAYER_RADIUS;
        hitWall = new THREE.Vector3(0, 0, 1);
      }
      player.vel.z = 0;
    }
    box.min = {x: player.pos.x - PLAYER_RADIUS, y: player.pos.y, z: player.pos.z - PLAYER_RADIUS};
    box.max = {x: player.pos.x + PLAYER_RADIUS, y: player.pos.y + PLAYER_HEIGHT, z: player.pos.z + PLAYER_RADIUS};
  }
  return hitWall;
}

function tryAttack(){
  if (player.attackCd > 0 || player.dead) return;
  player.attackT = 0.35;
  player.attackCd = 0.5;
  spawnSlashEffect();
  // check enemies
  const forward = new THREE.Vector3(Math.sin(player.yaw), 0, Math.cos(player.yaw));
  for (const e of enemies){
    if (!e.userData.alive) continue;
    const toE = new THREE.Vector3().subVectors(e.root.position, player.pos);
    toE.y = 0;
    const dist = toE.length();
    if (dist < 2.5){
      toE.normalize();
      const dot = forward.dot(toE);
      if (dot > 0.3){
        e.userData.hp--;
        e.userData.hurt = 0.3;
        spawnSparks(e.root.position.clone().setY(e.root.position.y + 1));
        if (e.userData.hp <= 0){
          e.userData.alive = false;
          player.score += 100;
          e.root.rotation.z = Math.PI/2;
          setTimeout(() => e.root.visible = false, 800);
        } else {
          player.score += 20;
        }
        updateHUD();
      }
    }
  }
}

const effects = [];
function updateEffects(dt){
  for (let i = effects.length - 1; i >= 0; i--){
    const e = effects[i];
    if (e.userData.rot){
      e.rotation.z += dt * 15;
      e.scale.setScalar(1 + (0.3 - e.userData.life) * 2);
    } else {
      e.userData.vel.y -= 15 * dt;
      e.position.addScaledVector(e.userData.vel, dt);
    }
    e.userData.life -= dt;
    if (e.userData.life <= 0){
      scene.remove(e);
      e.geometry?.dispose();
      effects.splice(i, 1);
    } else if (e.material && e.material.opacity !== undefined){
      e.material.opacity = Math.max(0, e.userData.life * 2);
    }
  }
}

function updatePlayer(dt){
  if (player.dead){
    player.deathT += dt;
    if (player.deathT > 1.5){
      gameState = 'dead';
      document.getElementById('over').classList.remove('hide');
    }
    return;
  }
  
  if (player.invuln > 0) player.invuln -= dt;
  if (player.attackCd > 0) player.attackCd -= dt;
  if (player.attackT > 0) player.attackT -= dt;
  if (player.wallJumpLock > 0) player.wallJumpLock -= dt;
  
  // INPUT
  let ix = 0, iz = 0;
  if (keys['ArrowLeft'] || keys['KeyA']) ix -= 1;
  if (keys['ArrowRight'] || keys['KeyD']) ix += 1;
  if (keys['ArrowUp'] || keys['KeyW']) iz -= 1;
  if (keys['ArrowDown'] || keys['KeyS']) iz += 1;
  if (touch.mx) ix += touch.mx;
  if (touch.my) iz += touch.my;
  const il = Math.hypot(ix, iz);
  if (il > 1){ ix /= il; iz /= il; }
  
  // Convert input to world-space using camera yaw
  const camForward = new THREE.Vector3(
    player.pos.x - camera.position.x, 0, player.pos.z - camera.position.z
  ).normalize();
  const camRight = new THREE.Vector3(-camForward.z, 0, camForward.x);
  
  const wx = camRight.x * ix + camForward.x * (-iz);
  const wz = camRight.z * ix + camForward.z * (-iz);
  
  // WALL RUN detection
  let touchingWall = null;
  const wallCheckDirs = [
    {d: new THREE.Vector3(1, 0, 0), r: new THREE.Vector3(0, 0, -1)},
    {d: new THREE.Vector3(-1, 0, 0), r: new THREE.Vector3(0, 0, 1)},
    {d: new THREE.Vector3(0, 0, 1), r: new THREE.Vector3(1, 0, 0)},
    {d: new THREE.Vector3(0, 0, -1), r: new THREE.Vector3(-1, 0, 0)}
  ];
  const probe = getAABB(player.pos);
  const pd = 0.25;
  for (const wc of wallCheckDirs){
    const check = {
      min: {x: probe.min.x + wc.d.x * pd, y: probe.min.y + 0.3, z: probe.min.z + wc.d.z * pd},
      max: {x: probe.max.x + wc.d.x * pd, y: probe.max.y - 0.3, z: probe.max.z + wc.d.z * pd}
    };
    for (const c of colliders){
      if (c.type !== 'wall' && c.type !== 'box') continue;
      if (overlap(check, c)){
        touchingWall = {normal: wc.d.clone(), right: wc.r.clone()};
        break;
      }
    }
    if (touchingWall) break;
  }
  
  const isAirborne = !player.onGround;
  const inputMag = Math.hypot(wx, wz);
  const facingWall = touchingWall && (
    wx * -touchingWall.normal.x + wz * -touchingWall.normal.z > 0.15
  );
  
  // enter wall run
  if (isAirborne && facingWall && player.wallJumpLock <= 0 && player.vel.y < 4){
    player.onWall = true;
    player.wallNormal.copy(touchingWall.normal);
    // direction along wall
    const along = touchingWall.right;
    const dot = wx * along.x + wz * along.z;
    player.wallDir = Math.sign(dot) || 1;
  }
  
  if (player.onWall && !touchingWall){
    player.onWall = false;
  }
  if (player.onGround){
    player.onWall = false;
  }
  
  // VELOCITY
  if (player.onWall){
    // wall-running: move along wall
    const target = new THREE.Vector3(
      touchingWall.right.x * player.wallDir * WALLRUN_SPEED,
      0,
      touchingWall.right.z * player.wallDir * WALLRUN_SPEED
    );
    player.vel.x += (target.x - player.vel.x) * Math.min(1, dt * 10);
    player.vel.z += (target.z - player.vel.z) * Math.min(1, dt * 10);
    // reduce gravity
    player.vel.y -= GRAV * dt * 0.25;
    // stick to wall
    const stick = 0.15;
    player.vel.x += touchingWall.normal.x * -stick;
    player.vel.z += touchingWall.normal.z * -stick;
  } else {
    // normal movement
    const targetVx = wx * MOVE_SPEED;
    const targetVz = wz * MOVE_SPEED;
    const acc = player.onGround ? 18 : 6;
    player.vel.x += (targetVx - player.vel.x) * Math.min(1, acc * dt);
    player.vel.z += (targetVz - player.vel.z) * Math.min(1, acc * dt);
    player.vel.y -= GRAV * dt;
  }
  
  if (player.vel.y < -MAX_FALL) player.vel.y = -MAX_FALL;
  
  // JUMP
  if (player.coyote > 0) player.coyote -= dt * 60;
  if (player.jumpBuf > 0) player.jumpBuf -= dt * 60;
  
  if (player.jumpBuf > 0){
    if (player.onWall){
      // wall jump: push away from wall
      player.vel.y = JUMP_V;
      player.vel.x = touchingWall.normal.x * 8;
      player.vel.z = touchingWall.normal.z * 8;
      // slight forward boost
      player.vel.x += wx * 3;
      player.vel.z += wz * 3;
      player.onWall = false;
      player.jumpBuf = 0;
      player.wallJumpLock = 0.3;
      spawnDust(player.pos.clone());
    } else if (player.onGround || player.coyote > 0){
      player.vel.y = JUMP_V;
      player.onGround = false;
      player.coyote = 0;
      player.jumpBuf = 0;
      player.doubleJumpUsed = false;
    } else if (!player.doubleJumpUsed){
      player.vel.y = JUMP_V * 0.9;
      player.doubleJumpUsed = true;
      player.jumpBuf = 0;
      spawnSlashEffect();
    }
  }
  
  // MOVE + COLLIDE
  player.onGround = false;
  player.pos.x += player.vel.x * dt;
  resolveCollisions('x', player.vel.x * dt);
  player.pos.z += player.vel.z * dt;
  resolveCollisions('z', player.vel.z * dt);
  player.pos.y += player.vel.y * dt;
  resolveCollisions('y', player.vel.y * dt);
  
  // fall death
  if (player.pos.y < -15){
    damagePlayer(true);
  }
  
  // LOOK DIRECTION
  if (inputMag > 0.1 && !player.onWall){
    const targetYaw = Math.atan2(wx, wz);
    let d = targetYaw - player.yaw;
    while (d > Math.PI) d -= Math.PI * 2;
    while (d < -Math.PI) d += Math.PI * 2;
    player.yaw += d * Math.min(1, 15 * dt);
  }
  
  // POSE
  const speed2 = Math.hypot(player.vel.x, player.vel.z);
  let newPose = 'idle';
  if (player.attackT > 0) newPose = 'attack';
  else if (player.onWall) newPose = 'wallrun';
  else if (!player.onGround && player.vel.y > 2) newPose = 'jump';
  else if (!player.onGround && player.vel.y < -2) newPose = 'fall';
  else if (speed2 > 5.5) newPose = 'run';
  else if (speed2 > 0.5) newPose = 'walk';
  else newPose = 'idle';
  
  if (newPose !== player.pose){
    player.pose = newPose;
    player.runTime = 0;
  }
  
  // animate
  if (player.pose === 'walk' || player.pose === 'run'){
    player.runTime += dt * 6;
  } else if (player.pose === 'attack'){
    player.runTime = (0.35 - player.attackT) * 3;
  } else if (player.pose === 'wallrun'){
    player.runTime += dt * 6;
  } else {
    player.runTime += dt;
  }
  
  poseCharacter(prince, player.pose, player.runTime, speed2);
  
  // update prince
  prince.root.position.copy(player.pos);
  prince.root.rotation.y = player.yaw;
  
  // CARROTS
  for (const c of carrots){
    if (c.userData.got) continue;
    c.userData.t += dt * 2;
    c.rotation.y += dt * 1.5;
    c.position.y = c.userData.baseY + Math.sin(c.userData.t) * 0.15;
    if (c.position.distanceTo(player.pos.clone().setY(player.pos.y + 0.9)) < 1.3){
      c.userData.got = true;
      c.visible = false;
      player.carrotCount++;
      player.score += 25;
      updateHUD();
    }
  }
  
  // GOAL
  if (goal.position.distanceTo(player.pos.clone().setY(player.pos.y + 1)) < 2.5){
    gameState = 'won';
    document.getElementById('win').classList.remove('hide');
  }
  
  // ENEMIES
  for (const e of enemies){
    if (!e.userData.alive) continue;
    const ud = e.userData;
    ud.t += dt;
    ud.hurt = Math.max(0, ud.hurt - dt);
    
    const toPlayer = new THREE.Vector3().subVectors(player.pos, e.root.position);
    toPlayer.y = 0;
    const distToPlayer = toPlayer.length();
    
    if (distToPlayer < 6){
      // chase
      toPlayer.normalize();
      const speed = distToPlayer > 2 ? 3.5 : 0;
      e.root.position.x += toPlayer.x * speed * dt;
      e.root.position.z += toPlayer.z * speed * dt;
      e.root.rotation.y = Math.atan2(toPlayer.x, toPlayer.z);
      poseCharacter(e, distToPlayer > 1.5 ? 'run' : 'idle', ud.t, 1);
      
      // attack
      ud.attackCd -= dt;
      if (distToPlayer < 2.2 && ud.attackCd <= 0 && player.invuln <= 0){
        ud.attackCd = 1.2;
        poseCharacter(e, 'attack', 0, 1);
        setTimeout(() => damagePlayer(false), 250);
      }
    } else {
      // patrol
      e.root.position.x = ud.base.x + Math.sin(ud.t * 0.6) * ud.patrol;
      e.root.rotation.y = Math.cos(ud.t * 0.6) > 0 ? Math.PI/2 : -Math.PI/2;
      poseCharacter(e, 'walk', ud.t * 1.5, 1);
    }
  }
}

function damagePlayer(fell){
  if (player.invuln > 0 && !fell) return;
  if (player.dead) return;
  player.hp--;
  player.invuln = 1.2;
  updateHUD();
  poseCharacter(prince, 'hurt', 0, 1);
  setTimeout(() => {
    if (player.hp <= 0 || fell){
      player.dead = true;
      player.deathT = 0;
      poseCharacter(prince, 'hurt', 0, 1);
    }
  }, 300);
}

/* ═══════════ CAMERA UPDATE ═══════════ */
function updateCamera(dt){
  const targetY = player.pos.y + 2.5;
  const back = 6;
  const up = 3.2;
  
  // Camera behind player based on yaw
  const offsetX = -Math.sin(player.yaw) * back;
  const offsetZ = -Math.cos(player.yaw) * back;
  
  camTargetPos.set(
    player.pos.x + offsetX,
    targetY + up,
    player.pos.z + offsetZ
  );
  
  // smooth follow
  camOffset.lerp(camTargetPos, 1 - Math.pow(0.001, dt));
  camera.position.copy(camOffset);
  
  camLook.lerp(
    new THREE.Vector3(player.pos.x, player.pos.y + 1.2, player.pos.z),
    1 - Math.pow(0.001, dt)
  );
  camera.lookAt(camLook);
  
  // sun follows
  sun.position.set(player.pos.x + 30, player.pos.y + 60, player.pos.z + 25);
  sun.target.position.copy(player.pos);
  sun.target.updateMatrixWorld();
}

/* ═══════════ MAIN LOOP ═══════════ */
let last = performance.now();
function loop(now){
  requestAnimationFrame(loop);
  const dt = Math.min(0.04, (now - last) / 1000);
  last = now;
  
  // torches flicker
  for (const t of torches){
    const f = 1 + Math.sin(now * 0.008 + t.offset) * 0.15 + Math.sin(now * 0.021 + t.offset * 2) * 0.08;
    t.mesh.scale.set(0.8 * f, 1.4 * f, 0.8 * f);
    t.light.intensity = 4 * f;
  }
  
  // goal ring spin
  if (goalRing){
    goalRing.rotation.y += dt * 1.5;
    goalRing.rotation.x = Math.sin(now * 0.001) * 0.3;
  }
  if (goalDisc){
    goalDisc.rotation.y -= dt * 0.8;
  }
  
  if (gameState === 'play'){
    updatePlayer(dt);
  } else if (gameState === 'menu'){
    // menu camera
    const t = now * 0.0004;
    const menuTarget = new THREE.Vector3(Math.cos(t) * 14, 8, Math.sin(t) * 14);
    camera.position.lerp(menuTarget, 0.05);
    camera.lookAt(0, 3, 0);
    // idle pose
    if (!prince.root.parent) scene.add(prince.root);
    prince.root.position.set(0, 1.5, 8);
    prince.root.rotation.y = Math.PI;
    poseCharacter(prince, 'idle', now * 0.001, 0);
  } else {
    // dead or won: freeze frame but keep idle anim
    poseCharacter(prince, player.dead ? 'hurt' : 'idle', now * 0.001, 0);
  }
  
  updateEffects(dt);
  
  // stars twinkle
  if (stars){
    stars.rotation.y += dt * 0.005;
  }
  
  composer.render();
}
